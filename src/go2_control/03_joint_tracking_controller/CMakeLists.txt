cmake_minimum_required(VERSION 3.8)
project(go2_joint_tracking_controller)
find_package(ament_cmake REQUIRED)
find_package(controller_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(realtime_tools REQUIRED)
find_package(sensor_msgs REQUIRED)
add_library(go2_balance_effort_controller SHARED src/balance_effort_controller.cpp)
target_compile_features(go2_balance_effort_controller PUBLIC cxx_std_17)
ament_target_dependencies(go2_balance_effort_controller controller_interface pluginlib realtime_tools sensor_msgs)
pluginlib_export_plugin_description_file(controller_interface balance_plugins.xml)
install(TARGETS go2_balance_effort_controller LIBRARY DESTINATION lib)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_package()
