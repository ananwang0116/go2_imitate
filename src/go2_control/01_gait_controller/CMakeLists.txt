cmake_minimum_required(VERSION 3.8)
project(go2_gait_controller)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(urdf REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(gazebo_msgs REQUIRED)
find_package(controller_manager_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
add_executable(champ_controller src/champ_controller.cpp)
target_include_directories(champ_controller PRIVATE include vendor/champ)
target_link_libraries(champ_controller Eigen3::Eigen)
target_compile_options(champ_controller PRIVATE -O2)
ament_target_dependencies(champ_controller rclcpp urdf geometry_msgs sensor_msgs std_msgs trajectory_msgs gazebo_msgs controller_manager_msgs)
add_executable(reference_trot_controller src/reference_trot_controller.cpp)
target_include_directories(reference_trot_controller PRIVATE include vendor/champ)
target_link_libraries(reference_trot_controller Eigen3::Eigen)
target_compile_options(reference_trot_controller PRIVATE -O2)
ament_target_dependencies(reference_trot_controller rclcpp urdf geometry_msgs sensor_msgs std_msgs trajectory_msgs gazebo_msgs controller_manager_msgs)
install(TARGETS reference_trot_controller DESTINATION lib/${PROJECT_NAME})
add_executable(leg_odometry src/leg_odometry.cpp)
target_include_directories(leg_odometry PRIVATE include vendor/champ)
target_link_libraries(leg_odometry Eigen3::Eigen)
target_compile_options(leg_odometry PRIVATE -O2)
ament_target_dependencies(leg_odometry rclcpp urdf geometry_msgs sensor_msgs std_msgs gazebo_msgs nav_msgs tf2_ros)
install(TARGETS leg_odometry DESTINATION lib/${PROJECT_NAME})
install(TARGETS champ_controller DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES vendor/UPSTREAM.md vendor/champ/LICENSE DESTINATION share/${PROJECT_NAME}/vendor)
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_kinematics test/test_kinematics.cpp)
  target_include_directories(test_kinematics PRIVATE include vendor/champ)
  target_link_libraries(test_kinematics Eigen3::Eigen)
  target_compile_options(test_kinematics PRIVATE -O2)
  ament_target_dependencies(test_kinematics urdf)
  add_test(NAME go2_champ_kinematics COMMAND test_kinematics ${CMAKE_CURRENT_SOURCE_DIR}/../../go2_description/urdf/go2_description.urdf)
endif()
ament_package()
