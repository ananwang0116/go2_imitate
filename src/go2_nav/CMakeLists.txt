cmake_minimum_required(VERSION 3.8)
project(go2_nav)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(go2_msgs REQUIRED)
find_package(yaml-cpp REQUIRED)

add_executable(mot_nav src/mot_nav.cpp)
target_include_directories(mot_nav PRIVATE include)
ament_target_dependencies(mot_nav rclcpp tf2 tf2_ros geometry_msgs nav_msgs sensor_msgs diagnostic_msgs go2_msgs)
target_link_libraries(mot_nav yaml-cpp)
install(TARGETS mot_nav DESTINATION lib/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_nav_logic test/test_nav_logic.cpp)
  target_include_directories(test_nav_logic PRIVATE include)
  target_link_libraries(test_nav_logic yaml-cpp)
endif()

ament_package()
